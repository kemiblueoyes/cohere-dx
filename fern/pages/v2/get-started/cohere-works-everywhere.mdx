Cohere's SDKs support Cohere platform, Amazon Bedrock, SageMaker, Azure AI Foundry, OCI, and private deployment environments. The client class and initialization pattern differ by deployment, but the rest of your code stays the same, so switching environments requires only minimal changes.

For SDK installation, see [SDKs](/docs/sdks).

## Feature support

The Cohere platform supports the full set of available features. Cloud and private deployment options support subsets of these features. See each platform's documentation for details.

| Feature                                                              | Cohere Platform | Bedrock       | Sagemaker     | Azure         | OCI           | Private Deployment |
| -------------------------------------------------------------------- | --------------- | ------------- | ------------- | ------------- | ------------- | ------------------ |
| chat_stream                                                          | ✅               | ✅             | ✅             | ✅             | ✅             | ✅                  |
| chat                                                                 | ✅               | ✅             | ✅             | ✅             | ✅             | ✅                  |
| embed                                                                | ✅               | ✅             | ✅             | ✅             | ✅             | ✅                  |
| rerank                                                               | ✅               | ✅             | ✅             | ✅             | ⬜️            | ✅                  |
| tokenize                                                             | ✅               | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*)      |
| detokenize                                                           | ✅               | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*) | ✅ (*offline*)      |

**Offline** refers to [tokenization](/docs/tokens-and-tokenizers#caching-and-optimization) that runs locally without making a network call to Cohere's API. The tokenizer config is downloaded and cached on your local machine, so subsequent tokenization happens without hitting Cohere's servers. In the Cohere Python SDK only, these can be accessed by setting `offline=False`.

## Code examples

The following examples show how to initialize a client and make an API call for all deployment options.

<Note>
  The Cohere v2 API is not yet supported for Bedrock, SageMaker, and Azure. OCI supports the v2 API via `OciClientV2`. The code examples shown for Bedrock, SageMaker, and Azure use the v1 API.
</Note>

```python
import cohere

co = cohere.ClientV2("COHERE_API_KEY")

res = co.chat(
    model="command-a-plus-05-2026",
    messages=[
        {"role": "user", "content": "Who discovered gravity?"},
        {
            "role": "assistant",
```

```go
package main

import (
	"context"
	"log"

	cohere "github.com/cohere-ai/cohere-go/v2"
	client "github.com/cohere-ai/cohere-go/v2/client"
)

func main() {
    co := client.NewClient(client.WithToken("COHERE_API_KEY"))

	resp, err := co.V2.Chat(
		context.TODO(),
		&cohere.V2ChatRequest{
			Model: "command-a-plus-05-2026",
			Messages: cohere.ChatMessages{
				{
					Role: "user",
					User: &cohere.UserMessageV2{Content: &cohere.UserMessageV2Content{
						String: "Who discovered gravity?",
					}},
				},
				{
					Role: "assistant",
					Assistant: &cohere.AssistantMessage{Content: &cohere.AssistantMessageV2Content{
						String: "The man who is widely credited with discovering gravity is Sir Isaac Newton.",
					}},
				},
				{
					Role: "user",
					User: &cohere.UserMessageV2{Content: &cohere.UserMessageV2Content{
						String: "What year was he born?",
					}},
				},
			},
		},
	)

	if err != nil {
		log.Fatal(err)
	}

	log.Printf("%+v", resp)
}
```
```java
package chatv2post;

import com.cohere.api.Cohere;
import com.cohere.api.resources.v2.requests.V2ChatRequest;
import com.cohere.api.types.*;
import java.util.List;

public class Default {
    public static void main(String[] args) {
        Cohere cohere = Cohere.builder().token("COHERE_API_KEY").clientName("snippet").build();

        ChatResponse response =
            cohere.v2()
                .chat(
                    V2ChatRequest.builder()
                        .model("command-a-plus-05-2026")
                        .messages(
                            List.of(
                                ChatMessageV2.user(
                                    UserMessage.builder().content(
                                        UserMessageContent.of("Who discovered gravity?")).build()),
                                ChatMessageV2.assistant(
                                    AssistantMessage.builder().content(
                                        AssistantMessageV2Content.of(
                                            "The man who is widely credited with discovering gravity is Sir Isaac Newton.")).build()),
                                ChatMessageV2.user(
                                    UserMessage.builder().content(
                                        UserMessageContent.of("What year was he born?")).build())
                            )).build());

        System.out.println(response);
    }
}
```

```java
import com.cohere.api.Cohere;
import com.cohere.api.requests.ChatRequest;
import com.cohere.api.types.ChatMessage;
import com.cohere.api.types.Message;
import com.cohere.api.types.NonStreamedChatResponse;
import java.util.List;

public class ChatPost {
    public static void main(String[] args) {
        Cohere cohere = Cohere.builder().environment(Environment.custom("https://Cohere-command-r-plus-phulf-serverless.eastus2.inference.ai.azure.com/v1")).token("AZURE_INFERENCE_CREDENTIAL").clientName("snippet").build();

    }
```
